# PHASE 4: TEST MT5 TERMINAL CONNECTIVITY

Create:

`scripts/test_connection.py`

Use the following code:

```python
import MetaTrader5 as mt5
import sys

print("=" * 60)
print("MT5 TERMINAL CONNECTIVITY TEST")
print("=" * 60)

# Initialize the currently running MT5 terminal.

if not mt5.initialize():
    print("FAILED: MT5 initialization")

    print("Error:", mt5.last_error())

    sys.exit(1)

print("\nSUCCESS: MT5 initialized")

# Retrieve terminal information.

terminal = mt5.terminal_info()

if terminal is None:
    print("FAILED: Cannot retrieve terminal information")
    print(mt5.last_error())

    mt5.shutdown()
    sys.exit(1)

print("\nTERMINAL INFORMATION")

print("Connected:", terminal.connected)
print("Company:", terminal.company)
print("Name:", terminal.name)
print("Path:", terminal.path)
print("Data Path:", terminal.data_path)
print("Build:", terminal.build)

# Confirm broker connectivity.

if not terminal.connected:
    print("\nFAILED: Terminal is not connected to broker")

    mt5.shutdown()
    sys.exit(1)

# Retrieve account information.

account = mt5.account_info()

if account is None:
    print("FAILED: Cannot retrieve account information")
    print(mt5.last_error())

    mt5.shutdown()
    sys.exit(1)

print("\nSUCCESS: ACCOUNT INFORMATION AVAILABLE")

print("Login:", account.login)
print("Broker:", account.company)
print("Server:", account.server)
print("Currency:", account.currency)
print("Leverage:", account.leverage)
print("Balance:", account.balance)
print("Equity:", account.equity)
print("Free Margin:", account.margin_free)

print("\nACCOUNT MODE")

trade_modes = {
    0: "DEMO",
    1: "CONTEST",
    2: "REAL"
}

margin_modes = {
    0: "RETAIL NETTING",
    1: "EXCHANGE",
    2: "RETAIL HEDGING"
}

print(
    "Account Type:",
    trade_modes.get(account.trade_mode, "UNKNOWN")
)

print(
    "Margin Mode:",
    margin_modes.get(account.margin_mode, "UNKNOWN")
)

print("\nTRADING PERMISSIONS")

print("Trading Allowed:", account.trade_allowed)
print("Expert Trading Allowed:", account.trade_expert)

print("\nMT5 CONNECTION TEST COMPLETED")

mt5.shutdown()
```

Run:

```powershell
python scripts/test_connection.py
```

Expected results:

- MT5 initialization successful.
- Terminal connected = True.
- Account information retrieved.
- Broker identified.
- Correct server identified.
- Account type identified as DEMO.
- Currency and leverage retrieved.
- Margin mode identified.

Important:

Account information and terminal connectivity are separate checks.

A successful initialize() alone does not prove that the broker connection is active.

# PHASE 5: AUTOMATIC ACCOUNT DISCOVERY

Create:

`scripts/account_discovery.py`

Use this implementation:

```python
import MetaTrader5 as mt5
import json
import os

if not mt5.initialize():
    raise RuntimeError(
        f"MT5 initialization failed: {mt5.last_error()}"
    )

try:
    account = mt5.account_info()
    terminal = mt5.terminal_info()

    if account is None or terminal is None:
        raise RuntimeError(
            f"Unable to retrieve MT5 information: {mt5.last_error()}"
        )

    trade_modes = {
        0: "DEMO",
        1: "CONTEST",
        2: "REAL"
    }

    margin_modes = {
        0: "RETAIL NETTING",
        1: "EXCHANGE",
        2: "RETAIL HEDGING"
    }

    report = {
        "terminal": {
            "connected": terminal.connected,
            "company": terminal.company,
            "name": terminal.name,
            "build": terminal.build,
            "path": terminal.path
        },

        "account": {
            "login": account.login,
            "broker": account.company,
            "server": account.server,
            "type": trade_modes.get(
                account.trade_mode, "UNKNOWN"
            ),
            "currency": account.currency,
            "leverage": account.leverage,
            "margin_mode": margin_modes.get(
                account.margin_mode, "UNKNOWN"
            ),
            "balance": account.balance,
            "equity": account.equity,
            "margin": account.margin,
            "free_margin": account.margin_free,
            "trade_allowed": account.trade_allowed,
            "trade_expert": account.trade_expert
        }
    }

    os.makedirs("reports", exist_ok=True)

    with open(
        "reports/account_report.json",
        "w",
        encoding="utf-8"
    ) as file:

        json.dump(report, file, indent=4)

    print(json.dumps(report, indent=4))

    print("\nAccount discovery completed.")

finally:
    mt5.shutdown()
```

Run:

```powershell
python scripts/account_discovery.py
```

The generated report must not contain passwords or authentication tokens.

Mask the login in any externally shared report.

# PHASE 6: SYMBOL DISCOVERY

We need to discover the actual Gold instrument supported by the client's broker.

It may be named:

- XAUUSD
- XAUUSDm
- GOLD
- Another broker-specific symbol.

Create:

`scripts/symbol_discovery.py`

Code:

```python
import MetaTrader5 as mt5

if not mt5.initialize():
    raise RuntimeError(mt5.last_error())

try:

    symbols = mt5.symbols_get()

    if symbols is None:
        raise RuntimeError(mt5.last_error())

    print("\nGOLD SYMBOL DISCOVERY")

    gold_symbols = [
        symbol.name
        for symbol in symbols
        if (
            "XAU" in symbol.name.upper()
            or "GOLD" in symbol.name.upper()
        )
    ]

    if not gold_symbols:
        print("No Gold instruments discovered.")

    for symbol_name in gold_symbols:

        print("\n" + "=" * 50)

        info = mt5.symbol_info(symbol_name)

        if info is None:
            continue

        print("Symbol:", info.name)

        print("Description:", info.description)

        print(
            "Contract Size:",
            info.trade_contract_size
        )

        print("Minimum Volume:", info.volume_min)

        print("Maximum Volume:", info.volume_max)

        print("Volume Step:", info.volume_step)

        print("Digits:", info.digits)

        print("Point:", info.point)

        print("Tick Size:", info.trade_tick_size)

        print("Tick Value:", info.trade_tick_value)

        print("Spread:", info.spread)

        print("Trade Mode:", info.trade_mode)

        print("Filling Mode:", info.filling_mode)

        print("Execution Mode:", info.trade_exemode)

finally:
    mt5.shutdown()
```

Run:

```powershell
python scripts/symbol_discovery.py
```

The objective is to determine the exact symbol and its trading specifications.

Do not assume all brokers have identical Gold contracts.

# PHASE 7: LIVE MARKET DATA TEST

Create:

`scripts/test_market_data.py`

Code:

```python
import MetaTrader5 as mt5
import time

if not mt5.initialize():
    raise RuntimeError(mt5.last_error())

try:

    symbols = mt5.symbols_get()

    gold_symbols = [
        s.name for s in symbols
        if (
            "XAU" in s.name.upper()
            or "GOLD" in s.name.upper()
        )
    ]

    if not gold_symbols:
        print("No Gold symbols found.")
        raise SystemExit(1)

    symbol = gold_symbols[0]

    print("Testing instrument:", symbol)

    if not mt5.symbol_select(symbol, True):
        raise RuntimeError(
            f"Could not select {symbol}: {mt5.last_error()}"
        )

    print("\nLIVE MARKET DATA")

    for i in range(10):

        tick = mt5.symbol_info_tick(symbol)

        if tick is None:
            print("Tick unavailable:", mt5.last_error())

        else:

            print(
                f"Bid: {tick.bid} | "
                f"Ask: {tick.ask} | "
                f"Spread: {tick.ask - tick.bid:.5f} | "
                f"Time: {tick.time}"
            )

        time.sleep(1)

    print("\nMarket data test completed.")

finally:
    mt5.shutdown()
```

Run:

```powershell
python scripts/test_market_data.py
```

Expected:

- Correct Gold instrument discovered.
- Symbol successfully selected.
- Bid and Ask prices retrieved.
- Tick timestamps retrieved.
- Prices update when the market is active.

Do not interpret an unchanged price during a closed market as an API failure. Check the timestamp and market session.
